cmake_minimum_required(VERSION 3.16)
include(cmake/version.cmake)
project(EchoTrace VERSION ${ECHO_NUMERIC_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ECHO_BUILD_EDGE "Build the capture daemon and simulation backend" ON)
option(ECHO_RV1106 "Build the Rockchip/ALSA hardware capture backend" OFF)
option(ECHO_BUILD_MPP "Build the RK3576 MPP/RGA evidence extractor" OFF)
option(ECHO_BUILD_QT "Build the native Qt workbench" OFF)
option(ECHO_BUILD_LVGL "Build the Echo terminal UI" OFF)
option(ECHO_BUILD_VLM "Build the RKNN/RKLLM inference worker" OFF)
set(ECHO_DEPS_DIR "${CMAKE_SOURCE_DIR}/build/deps" CACHE PATH "Pinned dependency source directory")
set(ECHO_VENDOR_DIR "${CMAKE_SOURCE_DIR}/build/vendor" CACHE PATH "Adapted modules from existing projects")
if(NOT EXISTS "${ECHO_DEPS_DIR}/jsoncpp/src/lib_json/json_reader.cpp")
  message(FATAL_ERROR "Run python3 scripts/fetch_build_dependencies.py before configuring")
endif()
find_package(Threads REQUIRED)
add_library(echo_json STATIC ${ECHO_DEPS_DIR}/jsoncpp/src/lib_json/json_reader.cpp
  ${ECHO_DEPS_DIR}/jsoncpp/src/lib_json/json_value.cpp ${ECHO_DEPS_DIR}/jsoncpp/src/lib_json/json_writer.cpp)
target_include_directories(echo_json PUBLIC ${ECHO_DEPS_DIR}/jsoncpp/include)
target_compile_definitions(echo_json PUBLIC JSONCPP_STATICLIB)
add_library(echo_core native/src/core.cpp)
target_include_directories(echo_core PUBLIC native/include PRIVATE ${ECHO_DEPS_DIR}/picosha2)
target_link_libraries(echo_core PUBLIC echo_json Threads::Threads)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(echo_core PUBLIC stdc++fs)
endif()
if(EXISTS "${ECHO_VENDOR_DIR}/recording/recording.cc")
  add_library(echo_recording ${ECHO_VENDOR_DIR}/recording/recording.cc ${ECHO_VENDOR_DIR}/recording/mp4_mux.c)
  target_include_directories(echo_recording PUBLIC ${ECHO_VENDOR_DIR}/recording)
  target_link_libraries(echo_recording PUBLIC Threads::Threads)
endif()
if(ECHO_BUILD_EDGE)
  add_executable(echotrace-edge native/src/edge_main.cpp native/src/capture_sim.cpp native/src/media_lease.c)
  target_compile_definitions(echotrace-edge PRIVATE
    ECHOTRACE_VERSION="${ECHO_VERSION}"
    ECHOTRACE_SOURCE_REVISION="${ECHO_SOURCE_REVISION}")
  target_include_directories(echotrace-edge PRIVATE native/include)
  target_include_directories(echotrace-edge PRIVATE ${ECHO_DEPS_DIR}/websocketpp ${ECHO_DEPS_DIR}/asio/asio/include)
  target_compile_definitions(echotrace-edge PRIVATE ASIO_STANDALONE _WEBSOCKETPP_CPP11_STL_)
  target_link_libraries(echotrace-edge PRIVATE echo_core)
  if(ECHO_RV1106)
    include(cmake/rv1106.cmake)
  endif()
  install(TARGETS echotrace-edge DESTINATION bin)
endif()
if(ECHO_BUILD_QT)
  include(cmake/qt.cmake)
endif()
if(ECHO_BUILD_LVGL)
  include(cmake/lvgl.cmake)
endif()
if(ECHO_BUILD_VLM)
  include(cmake/vlm.cmake)
endif()
enable_testing()
add_executable(echotrace-media-lease-test native/tests/media_lease_test.c native/src/media_lease.c)
target_include_directories(echotrace-media-lease-test PRIVATE native/include)
target_compile_options(echotrace-media-lease-test PRIVATE -UNDEBUG)
if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME media_lease COMMAND echotrace-media-lease-test)
endif()
add_executable(echotrace-core-test native/tests/core_test.cpp)
target_link_libraries(echotrace-core-test PRIVATE echo_core)
target_compile_options(echotrace-core-test PRIVATE -UNDEBUG)
add_test(NAME core COMMAND echotrace-core-test)
if(TARGET echo_recording)
  add_executable(echotrace-mux-test native/tests/mux_test.cpp)
  target_link_libraries(echotrace-mux-test PRIVATE echo_recording)
  target_compile_options(echotrace-mux-test PRIVATE -UNDEBUG)
endif()
